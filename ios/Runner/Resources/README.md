# iOS Notification Sounds
# 
# Place your custom notification sound files here.
# Supported formats: .aiff, .wav, .caf
# 
# File naming conventions:
# - Use lowercase letters and underscores
# - Include the file extension when referencing in code
# - Example: custom_notification.aiff, alert_sound.wav
# 
# Note:
# - Files must be added to the iOS project bundle
# - Sounds should be 30 seconds or less
# - For best compatibility, use .aiff format
# - Reference them in code with the full filename including extension
# - Example: sound: 'custom_notification.aiff'
